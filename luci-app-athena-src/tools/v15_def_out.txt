===99-zz-athena-redirect===
#!/bin/sh
# ============================================================
#  AthenaWrt — 首页直达定制登录界面 (uci-defaults，首启执行一次)
#  luci-base 的 /www/index.html 会把「/」重定向到 cgi-bin/luci/，
#  导致先弹 LuCI 默认主题登录页、登录后再弹定制 SPA 自己的登录（两次密码）。
#  这里覆写 /www/index.html，让「/」直接跳到定制 SPA，
#  使 AthenaWrt 自带的登录页成为唯一一次登录。
#  （/www 在 squashfs 上，首启写入 overlay，持久生效；用 uci-defaults 而非
#   包内 INSTALL 是为了避免与 luci-base 的 /www/index.html 文件冲突。）
# ============================================================

WWW_INDEX=/www/index.html

cat > "$WWW_INDEX" <<'EOF'
<!DOCTYPE html>
<html>
	<head>
		<meta http-equiv="Cache-Control" content="no-cache, no-store, must-revalidate" />
		<meta http-equiv="Pragma" content="no-cache" />
		<meta http-equiv="Expires" content="0" />
		<meta http-equiv="refresh" content="0; URL=luci-static/resources/view/athena/index.html" />
	</head>
	<body>
		<a href="luci-static/resources/view/athena/index.html">AthenaWrt</a>
	</body>
</html>
EOF

exit 0
===99-athena-custom===
#!/bin/sh
# ============================================================
#  AthenaWrt 定制 — 首启配置 (uci-defaults，仅执行一次)
#  主机名 / 时区 / NTP / root 密码 / SSH
#  无线品牌 SSID 见 99-zz-athena-wifi.sh（排序在 99-jdc-wifi 之后覆盖）
# ============================================================

# ---------- 1. 主机名 & 时区 (中国标准时间) ----------
uci set system.@system[0].hostname='Athena'
uci set system.@system[0].timezone='CST-8'
uci set system.@system[0].zonename='Asia/Shanghai'
uci set system.@system[0].log_size='128'
uci set system.@system[0].log_type='file'
uci set system.@system[0].conloglevel='8'

# ---------- 2. NTP 时间同步 (国内源) ----------
uci -q delete system.ntp
uci set system.ntp=timeserver
uci add_list system.ntp.server='ntp.aliyun.com'
uci add_list system.ntp.server='ntp.tencent.com'
uci add_list system.ntp.server='cn.pool.ntp.org'
uci set system.ntp.enabled='1'
uci set system.ntp.enable_server='0'

# ---------- 3. root 默认密码 = password (首次登录后请务必修改) ----------
( echo "password"; echo "password" ) | passwd root 2>/dev/null

# ---------- 4. 启用 SSH (dropbear) ----------
if uci -q get dropbear.@dropbear[0] >/dev/null 2>&1; then
  uci set dropbear.@dropbear[0].RootPasswordAuth='on'
  uci set dropbear.@dropbear[0].PasswordAuth='on'
  uci set dropbear.@dropbear[0].Port='22'
fi

uci commit system
uci commit dropbear 2>/dev/null

exit 0

#!/bin/sh
# JDC Athena 三频 WiFi 默认调优（首次启动执行一次）
# radio 映射：radio0=2.4G(QCN5022) / radio1=5.8G(QCN5052 2x2) / radio2=5.2G(QCN9024 4x4 PCIe)
# 5.2G 与 5.8G 均 band=5g，按 htmode 能力区分（160MHz=5.2G，80MHz=5.8G）
# 参数：SSID 三频区分 / 密码 password / US / WPA2-PSK(CCMP) / 信道带宽按社区推荐值

BASE_KEY='password'

for r in 0 1 2; do
    band=$(uci -q get "wireless.radio${r}.band")
    [ -z "$band" ] && continue

    case "$band" in
        2g)
            SSID='JDC-Athena-2.4G'
            CH='11'
            HT='HT20'
            ;;
        5g)
            cur_ht=$(uci -q get "wireless.radio${r}.htmode")
            case "$cur_ht" in
                *160*)
                    SSID='JDC-Athena-5.2G'
                    CH='44'
                    HT='HE160'
                    ;;
                *)
                    SSID='JDC-Athena-5.8G'
                    CH='149'
                    HT='HE80'
                    ;;
            esac
            ;;
        *)
            continue
            ;;
    esac

    uci -q set "wireless.radio${r}.channel=$CH"
    uci -q set "wireless.radio${r}.htmode=$HT"
    uci -q set "wireless.radio${r}.country=US"
    uci -q set "wireless.radio${r}.disabled=0"

    # 第一个 wifi-iface 使用三频 SSID
    iface=$(uci -q get "wireless.radio${r}.ifname" 2>/dev/null)
    if [ -z "$iface" ]; then
        # 找到该 radio 下的第一个 wifi-iface
        idx=$(uci show wireless | grep -oE "wireless\.@wifi-iface\[[0-9]+\].device='radio${r}'" | grep -oE '\[[0-9]+\]' | head -1 | tr -d '[]')
        [ -z "$idx" ] && idx=$(uci show wireless | grep -oE "wireless\.@wifi-iface\[[0-9]+\]" | head -1 | grep -oE '[0-9]+')
        [ -z "$idx" ] && continue
        uci -q set "wireless.@wifi-iface[$idx].ssid=$SSID"
        uci -q set "wireless.@wifi-iface[$idx].encryption=psk2+ccmp"
        uci -q set "wireless.@wifi-iface[$idx].key=$BASE_KEY"
    fi
done

uci commit wireless
exit 0

#!/bin/sh

pio_file_check() {
	local file valid

	file=$1
	valid=0

	json_init
	json_load_file $file 2> /dev/null

	if json_is_a slaac array; then
		local keys len

		json_select slaac
		json_get_keys keys

		len=$(echo $keys | wc -w)
		if [ $len -gt 0 ]; then
			valid=1
		fi
	fi

	echo $valid
}

pio_folder_migrate() {
	local piodir piofolder

	piofolder=$1
	piodir=$2

	[ -d $piofolder ] || return
	[ -d $piodir ] || mkdir -p $piodir

	for file in $piofolder/*; do
		[ -e "$file" ] || continue
		if [ $(pio_file_check $file) -eq 1 ]; then
			mv $file $piodir/odhcpd.pio.$(basename $file)
		fi
	done
}

#!/bin/sh
# ============================================================
#  AthenaWrt 最终无线品牌覆盖 (uci-defaults)
#  说明：wifi detect (mac80211.uc) 是无线配置的真正生成点，本脚本作为
#        双保险兜底——若未来时序变化导致本脚本在 wireless 配置生成后执行，
#        可再次覆盖 SSID/密码/国家码。
#  兼容 wifi-iface 具名 section (default_radioN) 与匿名 section (@wifi-iface[N])。
#
#  三频默认：
#    2.4G          -> Athena-2.4G
#    5.2G (HE160)  -> Athena-5G-Game   (QCN9074 4×4 电竞频段)
#    5.8G (HE80)   -> Athena-5G        (板载 2×2 影音频段)
#  密码统一 12345678，WPA2-PSK，国家码 CN
# ============================================================

BASE_KEY='12345678'

for r in 0 1 2; do
	band=$(uci -q get "wireless.radio${r}.band")
	[ -z "$band" ] && continue
	htmode=$(uci -q get "wireless.radio${r}.htmode")

	case "$band" in
		2g) SSID='Athena-2.4G' ;;
		5g)
			case "$htmode" in
				*160*) SSID='Athena-5G-Game' ;;
				*)     SSID='Athena-5G' ;;
			esac
			;;
		*) continue ;;
	esac

	# 国家码 CN（5.8G 149-165 / 5.2G 36-64 均可用）
	uci -q set "wireless.radio${r}.country=CN"
	uci -q set "wireless.radio${r}.disabled=0"

	# 找到 device=radio${r} 的 wifi-iface section（兼容具名 default_radioN 与匿名 @wifi-iface[N]）
	iface=$(uci -q show wireless 2>/dev/null | grep -E "\.device='radio${r}'$" | head -1 | sed "s/\.device=.*//")
	if [ -n "$iface" ]; then
		uci -q set "${iface}.ssid=$SSID"
		uci -q set "${iface}.encryption=psk2"
		uci -q set "${iface}.key=$BASE_KEY"
	fi
done

uci commit wireless 2>/dev/null

exit 0

. /lib/functions.sh

migrate_ports() {
	local config="$1"
	local type ports ifname

	config_get type "$config" type
	[ "$type" != "bridge" ] && return

	config_get ports "$config" ports
	[ -n "$ports" ] && return

	config_get ifname "$config" ifname
	[ -z "$ifname" ] && return

	for port in $ifname; do
		uci add_list network.$config.ports="$port"
	done
	uci delete network.$config.ifname
}

migrate_bridge() {
	local config="$1"
	local type ifname

	config_get type "$config" type
	[ "$type" != "bridge" ] && return

	config_get ifname "$config" ifname

	uci -q batch <<-EOF
		add network device
		set network.@device[-1].name='br-$config'
		set network.@device[-1].type='bridge'
	EOF
	for port in $ifname; do
		uci add_list network.@device[-1].ports="$port"
	done

	uci -q batch <<-EOF
		delete network.$config.type
		delete network.$config.ifname
		set network.$config.device='br-$config'
	EOF
}

config_load network
config_foreach migrate_ports device
config_foreach migrate_bridge interface
